  function prompt_mise() {
    local plugins=("${(@f)$(mise ls --current --offline 2>/dev/null | awk '!/\(symlink\)/ && $3!="~/.tool-versions" && $3!="~/.config/mise/config.toml" && $3!="(missing)" {if ($1) print $1, $2}')}")
    for plugin in ${(k)plugins}; do
      local parts=("${(@s/ /)plugin}")
      local tool=${(U)parts[1]}
      local version=${parts[2]}
      p10k segment -r -i "${tool}_ICON" -s $tool -t "$version"
    done
  }
}

#!/usr/bin/env zsh
mise() {
  if [ "$1" = "use" ]; then
    tools=$(command mise ls --installed | awk '{print $1}' | sort -u)


    if [ "$2" = "-g" ]; then
      global_flag="-g"
      shift
    else
      global_flag=""
    fi
    
    if [ -n "$2" ] && ! echo "$tools" | grep -qx "$2"; then
      command mise "$@"
      return
    fi


    if [ -z "$2" ]; then
      # List installed tools
      
      if [ -z "$tools" ]; then
        echo "⚠️ No tools installed."
        echo "👉 Install with: mise install <tool>@<version>"
        return 1
      fi

      if command -v fzf >/dev/null 2>&1; then
        tool=$(echo "$tools" | fzf --prompt="Select tool > " --height=20 --reverse)
      else
        echo "Installed tools:"
        echo "$tools"
        read -p "Choose tool: " tool
      fi

      [ -z "$tool" ] && { echo "❌ No tool selected."; return 1; }

    else
      tool="$2"

      echo "Using tool: $tool"
    fi

    # Only proceed if tool does not contain '@'
    if [[ "$tool" != *"@"* ]]; then
      # List installed versions for the tool
      versions=$(command mise ls "$tool" --installed  | awk '{print $2}' | sort -V)
      if ! echo "$versions" | grep -qx "system"; then
        versions="system"$'\n'"$versions"
      fi
      if [ -z "$versions" ]; then
        echo "⚠️ No versions installed for '$tool'."
        echo "👉 Install with: mise install ${tool}@<version>"
        return 1
      fi

      if command -v fzf >/dev/null 2>&1; then
        chosen=$(echo "$versions" | fzf --prompt="Select version > " --height=20 --reverse)
      else
        echo "Installed versions for $tool:"
        echo "$versions"
        read -p "Choose version: " chosen
      fi

      [ -n "$chosen" ] && command mise use $global_flag "$tool@$chosen" || echo "❌ No version selected."
    fi

  else
    command mise "$@"
  fi
}
